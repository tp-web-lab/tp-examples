segment AC = segment(A, C)

point D = intersection(bis, BC, branch=0)
segment AD = segment(A, D) color=red
polygon left = polygon(A, B, D) color=blue
polygon rightPart = polygon(A, D, C) color=orange
comment bd = length(B,D)
comment dc = length(D,C)
comment ab = length(AB)
comment ac = length(AC)

comment beta = angle(D,A,C)
comment sl = area(left)

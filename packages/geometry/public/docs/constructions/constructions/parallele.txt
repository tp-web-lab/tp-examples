point A = (-3, 0)
point B = (3, 0)
segment AB = segment(A, B)
point P = (1, 3)
line qaxis = line(A, B) hidden
circle qcircle = circle(P, through=A) style=dotted color=zinc
point qX = intersection(qaxis, qcircle, branch=1)
circle qma = circle(A, distance(A, qX)) style=dotted color=zinc
circle qmb = circle(qX, distance(A, qX)) style=dotted color=zinc
point qmC = intersection(qma, qmb, branch=0)
point qmD = intersection(qma, qmb, branch=1)
line qmperp = line(qmC, qmD) hidden
line qmbase = line(A, qX) hidden
point qmM = intersection(qmperp, qmbase, branch=0)
point H = intersection(qmperp, qaxis, branch=0)
circle cp = circle(P, through=H) hidden
point J = intersection(cp, qmperp, branch=0)
circle va = circle(H, distance(H, J)) style=dotted color=zinc
circle vb = circle(J, distance(H, J)) style=dotted color=zinc
point vC = intersection(va, vb, branch=0)
point vD = intersection(va, vb, branch=1)
line vperp = line(vC, vD) color=red
line vbase = line(H, J) hidden
point vM = intersection(vperp, vbase, branch=0)

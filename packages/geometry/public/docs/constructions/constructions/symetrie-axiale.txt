point A = (-3, 0)
point B = (3, 0)
segment AB = segment(A, B)
point P = (1, 3)
circle ca = circle(A, through=P) style=dotted color=zinc
circle cb = circle(B, through=P) style=dotted color=zinc
point Q = intersection(ca, cb, branch=1)
segment PQ = segment(P, Q) color=red
point H = intersection(PQ, AB, branch=0)
segment PH = segment(P, H) color=blue
segment HQ = segment(H, Q) color=blue
comment lp = length(PH)
comment lq = length(HQ)
comment droit = angle(PQ, AB)
